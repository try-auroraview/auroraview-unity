cmake_minimum_required(VERSION 3.24)
project(auroraview_unity LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "AuroraView Unity currently targets Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(WEBVIEW2_SDK "" CACHE PATH "Extracted Microsoft.Web.WebView2 NuGet package")
if(NOT EXISTS "${WEBVIEW2_SDK}/build/native/include/WebView2.h")
  message(FATAL_ERROR "Run vx just fetch-sdk or supply WEBVIEW2_SDK")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../Editor/WebAssets/vendor/event_bridge.js" BRIDGE_SCRIPT)
string(LENGTH "${BRIDGE_SCRIPT}" BRIDGE_LENGTH)
set(BRIDGE_OFFSET 0)
set(BRIDGE_CHUNKS "")
while(BRIDGE_OFFSET LESS BRIDGE_LENGTH)
  math(EXPR BRIDGE_REMAINING "${BRIDGE_LENGTH} - ${BRIDGE_OFFSET}")
  if(BRIDGE_REMAINING GREATER 4096)
    set(BRIDGE_PART_LENGTH 4096)
  else()
    set(BRIDGE_PART_LENGTH ${BRIDGE_REMAINING})
  endif()
  string(SUBSTRING "${BRIDGE_SCRIPT}" ${BRIDGE_OFFSET} ${BRIDGE_PART_LENGTH} BRIDGE_PART)
  string(APPEND BRIDGE_CHUNKS "  script.append(LR\"AVBRIDGE(${BRIDGE_PART})AVBRIDGE\");\n")
  math(EXPR BRIDGE_OFFSET "${BRIDGE_OFFSET} + ${BRIDGE_PART_LENGTH}")
endwhile()
configure_file(bridge.h.in "${CMAKE_CURRENT_BINARY_DIR}/bridge.h" @ONLY)
add_library(auroraview_unity SHARED view.cpp)
target_include_directories(auroraview_unity PRIVATE "${WEBVIEW2_SDK}/build/native/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(auroraview_unity PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(auroraview_unity PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(auroraview_unity PRIVATE "${WEBVIEW2_SDK}/build/native/x64/WebView2LoaderStatic.lib" ole32 user32 shlwapi version advapi32)
add_custom_command(TARGET auroraview_unity POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_SOURCE_DIR}/../Editor/Plugins/x86_64"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:auroraview_unity>" "${CMAKE_CURRENT_SOURCE_DIR}/../Editor/Plugins/x86_64/auroraview_unity.dll")
add_executable(auroraview_native_test smoke.cpp)
target_compile_definitions(auroraview_native_test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(auroraview_native_test PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(auroraview_native_test PRIVATE auroraview_unity user32)
